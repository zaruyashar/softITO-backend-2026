public static double Ortalama(double not,double not2)
{
double sonuc = (not + not2) / 2d;
return sonuc;
}


public static int Zam(int fiyat,int tip)
{
if (tip ==1)
{
fiyat += fiyat * 10 / 100;
return fiyat;
}
else if(tip==2)
{
fiyat += fiyat * 12 / 100;
return fiyat;
}
else
{
return fiyat + 10;
}
}

public static int indirim(int fiyat, int tip)
{
if (tip ==1)
{
fiyat -= fiyat * 10 / 100;
return fiyat;
}
else if (tip ==2)
{
fiyat -= fiyat * 12 / 100;
return fiyat;
}
else
{
return fiyat - 10;
}
}
