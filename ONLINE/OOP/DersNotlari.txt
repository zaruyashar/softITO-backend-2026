{
    Console.WriteLine("Tecrübeniz nekadar");

    if(mezun=="lisans" && deneyim > 5)
    {
        maas += maas*30 / 100;
        return maas;
    }
    else
    {
        maas += maas * 10 / 100;
        return maas;
    }

}


public double PrimHesap(double maas,double prim)
{
    if (maas > 80000)
    {
        maas += prim;
        maas += 15000;
        return maas;
    }
    else
    {
        maas += prim;
        maas += 12000;
        return maas;
    }
}

{

}
else
{
return 0;
